unique_ptr<NodoAvl> avl_reequilibrar(unique_ptr<NodoAvl>& raiz, int &total_rotacao) {
    avl_atualizar_altura(raiz);
    int balanceamento = avl_fb(raiz);

    if (balanceamento == +2 && avl_fb(raiz ->esq) >= 0) {
        return avl_rotacao_direita(raiz, total_rotacao);
    } else if (balanceamento == -2 && avl_fb(raiz->dir) <= 0) {
        return avl_rotacao_esquerda(raiz, total_rotacao);
    } else if (balanceamento == +2 && avl_fb(raiz->esq) < 0) {
        raiz->esq = avl_rotacao_esquerda(raiz->esq, total_rotacao);
        return avl_rotacao_direita(raiz, total_rotacao  );
    } else if (balanceamento == -2 && avl_fb(raiz->dir) > 0) {
        raiz->dir = avl_rotacao_direita(raiz->dir, total_rotacao);
        return avl_rotacao_esquerda(raiz, total_rotacao);
    }

    return std::move(raiz);
}

    if (!fila.inicio) {
        fila.fim = nullptr;

export module TAD_hash_cache;

using std::unique_ptr;
using std::make_unique;
using std::string;

const int M = 16; //tamanho da tabela hash

export{
    struct NodoHashCache {
        int cod_pedido;
        string receita;
        unique_ptr<NodoHashCache> prox;
    };

    struct HashCache_Pedidos{
        unique_ptr<NodoHashCache> tabela[M];
        int n = 0; //número de elementos na tabela
        int m = M; //tamanho da tabela

        int hit = 0; //número de acertos na busca
        int miss = 0; //número de falhas na busca

        int colisoes = 0; //número de colisões na inserção
    };

    //PROTÓTIPOS DAS FUNÇÕES INTERNAS
    int hash_multiplicativo(int chave, int m);
    void inserir_hash(HashCache_Pedidos *t, int cod_pedido, const string& receita);

    //retorna a receita se o código do pedido estiver na tabela hash, se não estiver, busca na avl, insere na tabela hash e retorna uma string vazia
    string buscar_hash(HashCache_Pedidos *t, int cod_pedido) {
        int i = hash_multiplicativo(cod_pedido, t->m);

        const NodoHashCache *p = t->tabela[i].get();
        while (p != nullptr) {
            if (p->cod_pedido == cod_pedido) {
                t->hit++;
                return p->receita;
            }
            p = p->prox.get();
        }

        t->miss++;
        return "";
    }

    void inserir_hash(HashCache_Pedidos *t, int cod_novo, const string& receita_nova) {
        int i = hash_multiplicativo(cod_novo, t->m);
        
        //cria um novo nodo e insere no início da lista encadeada na posição i da tabela hash
        auto novo = make_unique<NodoHashCache>();
        novo->cod_pedido = cod_novo;
        novo->receita = receita_nova;

        //verifica se já existe algum elemento na posição i da tabela hash, se sim, incrementa o número de colisões
        if (t->tabela[i] !=nullptr) {
            t->colisoes++;
        }

        novo->prox = std::move(t->tabela[i]); 
        t->tabela[i] = std::move(novo);
        t->n++;
    }

    float taxa_acerto(int hit, int miss) {
        if (hit + miss == 0) return 0.0f; //trata o caso de divisão por zero quando não houve buscas
        float resultado = (float)hit / (hit + miss);
        return resultado; //NA EXIBIÇÃO, MULTIPLICAR POR 100 PARA OBTER A TAXA EM PORCENTAGEM
    }

    void limpar_hash(HashCache_Pedidos *t) {
        if (t == nullptr) {
            return;
        }

        for (int i = 0; i < t->m; i++) {
            t->tabela[i] = nullptr;
        }

        t->n = 0;
        t->hit = 0;
        t->miss = 0;
        t->colisoes = 0;
    }

}

int hash_multiplicativo(int chave, int m) {
    double A  = 0.6180339887;
    double frac = chave * A - (int)(chave * A);
    return (int)(m * frac);
}

    };

    //PROTÓTIPOS DAS FUNÇÕES INTERNAS
    void redimensionar(ListaContigua_Historico& lista);

    void inicializar_listacontigua(ListaContigua_Historico& lista) {
        lista.tamanho = 0;
        lista.capacidade = 1000; 
        lista.elementos = make_unique<NodoListaContigua[]>(lista.capacidade);
    }

    bool inserir_listacontigua(ListaContigua_Historico& lista, NodoListaContigua novo, int posicao) {
        
        if (posicao < 0 || posicao > lista.tamanho) { //posição inválida
            return false;  
        }

        if (lista.tamanho >= lista.capacidade) { //verifico se ainda cabe, se não, redimensiono
            redimensionar(lista);
        }

        for (int i = lista.tamanho; i > posicao; i--) {
            lista.elementos[i] = std::move(lista.elementos[i - 1]);
        }

        lista.elementos[posicao] = novo;
        lista.tamanho++;
          
        return true;
    }

    NodoListaContigua* acessar_listacontigua(ListaContigua_Historico& lista, int posicao) {
        NodoListaContigua* resultado;

        if (posicao >= lista.tamanho || posicao < 0) {
            return nullptr;
        }

        resultado = &lista.elementos[posicao];

        return resultado;
    }

    int tamanho_listacontigua(const ListaContigua_Historico& l) {
        return l.tamanho;
    }

        ListaContigua_Historico copia;
        copia.tamanho = lista.tamanho;
        copia.capacidade = lista.capacidade;
        //preciso alocar a memoria para a cópia, se não, dará erro e não vai ter como armazenar
        copia.elementos = make_unique<NodoListaContigua[]>(copia.capacidade);

module;
#include <memory>
export module TAD_minheap;
using std::unique_ptr;
using std::make_unique;


export struct minHeap {
    static constexpr int capacidade = 8;
    int tamanho = 0;
    unique_ptr<int[]> pereciveis = make_unique<int[]>(capacidade);
};

void heapify(minHeap &heap, int i){
    int menor = i;
    int esquerda = 2 * i + 1;
    int direita = 2 * i + 2;

    if (esquerda < heap.tamanho && heap.pereciveis[esquerda] < heap.pereciveis[menor])
        menor = esquerda;

    if (direita < heap.tamanho && heap.pereciveis[direita] < heap.pereciveis[menor])
        menor = direita;

    if (menor != i) {
        std::swap(heap.pereciveis[i], heap.pereciveis[menor]);
        heapify(heap, menor);
    }
}

void siftup(minHeap &heap, int i) {
    while (i > 0 && heap.pereciveis[(i - 1) / 2] > heap.pereciveis[i]) {
        std::swap(heap.pereciveis[i], heap.pereciveis[(i - 1) / 2]);
        i = (i - 1) / 2;
    }
}

export bool removerHeap(minHeap &heap) {
    if (heap.tamanho == 0) {
        return false;
    }
    heap.pereciveis[0] = heap.pereciveis[heap.tamanho - 1];
    heap.tamanho--;
    heapify(heap, 0);
    return true;
}

export bool inserir(minHeap &heap, int valor) {
    if (heap.tamanho == heap.capacidade) {
        return false;
    }
    heap.tamanho++;
    heap.pereciveis[heap.tamanho - 1] = valor;
    siftup(heap, heap.tamanho - 1);
    return true;
}

export void descartarVencidos(minHeap &heap, int turnoAtual) {
    if (heap.tamanho == 0) {
        return;
    }
    while (heap.pereciveis[0] <= turnoAtual) {
        removerHeap(heap);
        if (heap.tamanho == 0) {
            break;
        }
    }
}

module;
#include <memory>
#include <string>
export module TAD_receita;
using std::string;
using std::unique_ptr;
using std::make_unique;

struct nodoReceita {
    char Ingrediente;
    unique_ptr<nodoReceita> prox;
};

export struct Receita {
    unique_ptr<nodoReceita> Ini;
};

/* FUNÇÕES INTERNAS */

export void inserirInicio(Receita &receita, char ingrediente) {
    auto novo = make_unique<nodoReceita>(ingrediente, std::move(receita.Ini));
    receita.Ini = std::move(novo);
}

export void inserirFim(Receita &receita, char ingrediente) {
    auto novo = make_unique<nodoReceita>(ingrediente, nullptr);
    nodoReceita* aux = receita.Ini.get();
    if (aux == nullptr) {
        receita.Ini = std::move(novo);
        return;
    }

    while (aux->prox != nullptr) {
        aux = aux->prox.get();
    }

    aux->prox = std::move(novo);
}

export void removerFim(Receita &receita) {
    nodoReceita* aux = receita.Ini.get();

    if (aux == nullptr) return;

    if (aux->prox == nullptr) {
        receita.Ini = nullptr;
        return;
    }

    while (aux->prox->prox != nullptr) {
        aux = aux->prox.get();
    }

    aux->prox = nullptr;
}

/* FUNÇÕES EXPORTADAS */

export Receita construirReceita(const string &resultadoBusca) {
    Receita receita;
    for (char ingrediente : resultadoBusca) {
        inserirFim(receita, ingrediente);
    }
    return receita;
}

export int tamanho(const Receita &receita) {
    const nodoReceita* aux = receita.Ini.get();
    if (aux == nullptr) return 0;
    int contador = 0;

    while (aux != nullptr) {
        contador++;
        aux = aux->prox.get();
    }

    return contador;
}

string stringInversaReceita(const unique_ptr<nodoReceita> &nodo) {
    string resultado = "";
    if(nodo->prox == nullptr) return resultado += nodo->Ingrediente;
    resultado = stringInversaReceita(nodo->prox);

    return resultado += nodo->Ingrediente;
}

export string stringReceita(const Receita &receita) {
    if (receita.Ini == nullptr) return "";
    return stringInversaReceita(receita.Ini);
}

        while (std::getline(arquivo, linha_cod)){
            remover_fim(linha_cod);
            
            if(linha_cod.empty()) continue;

            if (!std::getline(arquivo, linha_receita)) {
                resultado.pares_ignorados++;
                break;
            }
            remover_fim(linha_receita);

            int codigo;
            
            if (!converter_codigo(linha_cod, codigo) || !receita_valida(linha_receita)) {
                resultado.pares_ignorados++;
                continue;
            }
            
            avl.raiz = inserir_pedido(avl.raiz, {codigo, linha_receita, nullptr, nullptr, 0}, avl.total_insercao, avl.total_rotacao, avl.total_comparacao, avl.menor_cod, avl.maior_cod);
        }

        resultado.tempo_carregamento = timer_parar(&t);
        resultado.sucesso = true;
        return resultado;

    }

    struct ResultadoHistorico {
        bool ok = false;        // false se o arquivo existe mas nao pode ser aberto
        long entradas = 0;      // entradas carregadas na lista
        long ignoradas = 0;     // pares invalidos que foram pulados
    } ;

    ResultadoHistorico carregar_historico(const string& caminho, ListaContigua_Historico& lista) {
        ResultadoHistorico res;

        std::error_code ec;
        if (!std::filesystem::exists(caminho, ec)) {
            res.ok = true;      // primeira execucao: historico comeca vazio
            return res;
        }

        std::ifstream arq(caminho);
        if (!arq.is_open()) {
            std::cerr << "Erro: nao foi possivel abrir o historico '" << caminho << "'\n";
            return res;
        }

        string linha_nome, linha_pontos;
        while (std::getline(arq, linha_nome)) {
            remover_fim(linha_nome);
            if (linha_nome.empty()) continue;            // linha em branco

            if (!std::getline(arq, linha_pontos)) {      // nome sem pontuacao no fim
                res.ignoradas++;
                break;
            }
            remover_fim(linha_pontos);

            int pontuacao = 0;
            if (!converter_codigo(linha_pontos, pontuacao) || pontuacao < 0) {
                res.ignoradas++;
                continue;
            }

            NodoListaContigua novo{linha_nome, pontuacao};
            if (!inserir_listacontigua(lista, novo, tamanho_listacontigua(lista))) {
                res.ignoradas++;
                continue;
            }
            res.entradas++;
        }

        res.ok = true;
        return res;
    }

    bool anexar_historico(const std::string& caminho, const std::string& nome, int pontuacao) {
        std::string limpo;
        for (char c : nome) {
            if (c != '\r' && c != '\n') limpo += c;
        }
        remover_fim(limpo);
        if (limpo.empty() || pontuacao < 0) return false;

        bool precisa_quebra = false;
        {
            std::ifstream leitura(caminho, std::ios::binary | std::ios::ate);
            if (leitura.is_open() && leitura.tellg() > 0) {
                leitura.seekg(-1, std::ios::end);
                char ultimo = 0;
                leitura.get(ultimo);
                precisa_quebra = (ultimo != '\n');
            }
        }

        std::ofstream saida(caminho, std::ios::app);
        if (!saida.is_open()) {
            std::cerr << "Erro: nao foi possivel gravar no historico '" << caminho << "'\n";
            return false;
        }
        if (precisa_quebra) saida << '\n';
        saida << limpo << '\n' << pontuacao << '\n';
        return saida.good();
    }
}

void remover_fim(string& str) {
    while (!str.empty() && (str.back() == '\r' || str.back() == ' ' || str.back() == '\t' || str.back() == '\n')) {
        str.pop_back();
    }
}

bool converter_codigo(const string& str, int& codigo) {
    try {
        size_t pos;
        codigo = std::stoi(str, &pos);

        if (pos != str.length()) {
            return false; //caracteres não numéricos na string
        }

    } catch (const std::invalid_argument&) {
        return false; //a string não é um número válido

    } catch (const std::out_of_range&) {
        return false; //número fora do intervalo representável por um int

    }
    return true;
}

bool receita_valida(const string& receita) {
    return receita.size() >= 5 && receita.size() <= 7;
}

Receita sortearReceita(Avl_Pedidos &avl, HashCache_Pedidos &cache, Metricas &metricas) {
    uniform_int_distribution<> dis(avl.menor_cod, avl.maior_cod);
    int sorteado;
    string stringret;
    Timer t;
    
    do{
        sorteado = dis(gen);
        stringret = buscar_hash(&cache, sorteado);
        if(!stringret.empty()) return construirReceita(stringret);

        timer_iniciar(&t);
        stringret = buscar_pedido(avl.raiz, sorteado, avl.total_comparacao);
        metricas.avl_tempo_buscas += timer_parar(&t);
    }while(stringret.empty());

    inserir_hash(&cache, sorteado, stringret);
    return construirReceita(stringret);
}

bool montarIngrediente(EstadoJogo& jogo, char ing) {
    // Evita consumir estoque se o pao topo ('z') ja fechou o lanche
    string pilhaAtual = stringPilha(jogo.montagem);
    if (!pilhaAtual.empty() && pilhaAtual[0] == 'z') {

            if (!removerHeap(jogo.estoque.carne)) {

                return false;
            }
            break;
        case 'o':
            if (!removerHeap(jogo.estoque.ovo)) {
                jogo.feedback = "Sem ovo pronto na chapa! Aperte [2] para preparar.";
                return false;
            }
            break;
        case 'b':
            if (!removerHeap(jogo.estoque.bacon)) {
                jogo.feedback = "Sem bacon pronto na chapa! Aperte [3] para preparar.";
                return false;
            }
            break;
        case 'q':
            if (jogo.estoque.queijoPrato <= 0) {
                jogo.feedback = "Sem queijo prato! Aperte [4] para repor.";
                return false;
            }
            jogo.estoque.queijoPrato--;
            break;
        case 'd':
            if (jogo.estoque.queijoCheddar <= 0) {
                jogo.feedback = "Sem cheddar! Aperte [5] para repor.";
                return false;
            }
            jogo.estoque.queijoCheddar--;
            break;
        case 'e':
            if (jogo.estoque.cebola <= 0) {
                jogo.feedback = "Sem cebola! Aperte [6] para repor.";
                return false;
            }
            jogo.estoque.cebola--;
            break;
        case 't':
            if (jogo.estoque.tomate <= 0) {
                jogo.feedback = "Sem tomate! Aperte [7] para repor.";
                return false;
            }
            jogo.estoque.tomate--;
            break;
        case 'z':
            // Pao topo e livre (nao consome estoque)
            break;
        default:
            return false;
    }

    empilhar(jogo.montagem, ing);
    jogo.feedback = string("Ingrediente '") + ing + "' empilhado!";
    return true;
}

bool entregarPedido(EstadoJogo& jogo) {
    Cliente* atual = visualizarInicio(jogo.fila);
    if (atual == nullptr) {
        jogo.rodando = false;
        return false;
    }

    if (comparar(atual->pedido, jogo.montagem)) {
        int ganho = 100 + atual->paciencia * 10;
        jogo.pontuacao += ganho;
        jogo.clientesAtendidos++;

        desenfileirar(jogo.fila);
        iniciarMontagem(jogo.montagem);

        if (jogo.fila.tamanho == 0) {
            jogo.feedback = "Expediente concluido! Todos os clientes foram atendidos!";
            jogo.rodando = false;
        } else {
            jogo.feedback = "Pedido entregue com sucesso! +" + std::to_string(ganho) + " pts.";
        }
        return true;
    }

    jogo.feedback = "Pedido incorreto! Use [Backspace] para corrigir e tente novamente.";
    return true; // penalidade de 1 turno por entrega errada
}

void avancarTurno(EstadoJogo& jogo) {
    if (!jogo.rodando) return;

    jogo.turnoAtual++;
    descartarVencidos(jogo.estoque.carne, jogo.turnoAtual);
    descartarVencidos(jogo.estoque.ovo,   jogo.turnoAtual);
    descartarVencidos(jogo.estoque.bacon, jogo.turnoAtual);

    Cliente* atual = visualizarInicio(jogo.fila);
    if (atual != nullptr) {
        atual->paciencia--;
        if (atual->paciencia <= 0) {
            desenfileirar(jogo.fila);
            iniciarMontagem(jogo.montagem);
            jogo.clientesPerdidos++;
            jogo.feedback = "Cliente perdeu a paciencia e foi embora! ("
                          + std::to_string(jogo.clientesPerdidos) + "/"
                          + std::to_string(MAX_PERDIDOS) + ")";
        }
    }

    if (jogo.clientesPerdidos >= MAX_PERDIDOS || jogo.fila.tamanho == 0) {
        jogo.rodando = false;
    }
}

void processarTecla(EstadoJogo& jogo, char tecla) {

        case '5': gastouTurno = reporPerene(jogo, 'd'); break;
        case '6': gastouTurno = reporPerene(jogo, 'e'); break;
        case '7': gastouTurno = reporPerene(jogo, 't'); break;

        // Montar hambúrguer (Pilha)
        case 'c': case 'C': gastouTurno = montarIngrediente(jogo, 'c'); break;
        case 'o': case 'O': gastouTurno = montarIngrediente(jogo, 'o'); break;
        case 'b': case 'B': gastouTurno = montarIngrediente(jogo, 'b'); break;
        case 'q': case 'Q': gastouTurno = montarIngrediente(jogo, 'q'); break;
        case 'd': case 'D': gastouTurno = montarIngrediente(jogo, 'd'); break;
        case 'e': case 'E': gastouTurno = montarIngrediente(jogo, 'e'); break;
        case 't': case 'T': gastouTurno = montarIngrediente(jogo, 't'); break;
        case 'z': case 'Z': gastouTurno = montarIngrediente(jogo, 'z'); break;

        // Desfazer (Linux = 127, Windows = '\b')
        case 127:
        case '\b':
            gastouTurno = desfazerIngrediente(jogo);
            break;

        // Entregar (Linux = '\n', Windows = '\r')
        case '\n':
        case '\r':
            gastouTurno = entregarPedido(jogo);
            break;

        // Sair (ESC ou 0)
        case 27:
        case '0':
            jogo.rodando = false;
            return;

        default:
            return;
    }

    if (gastouTurno) {
        avancarTurno(jogo);
    }
}

export int executarGameplay(Avl_Pedidos& avl,
                            HashCache_Pedidos& cache,
                            Metricas& metricas,
                            const PacoteSprites& sprites) {
    EstadoJogo jogo;
    inicializarPartida(jogo, avl, cache, metricas);

    ativarTerminal();

    Timer timerAnim;
    timer_iniciar(&timerAnim);
    int tickAnim = 0;
    bool precisaRedesenhar = true;

    while (jogo.rodando) {
        // Anima sprites ~4 vezes por segundo (independente do teclado)
        if (timer_parar(&timerAnim) >= 0.25) {
            tickAnim++;
            timer_iniciar(&timerAnim);
            precisaRedesenhar = true;
        }

        if (precisaRedesenhar) {
            DadosTela dados = montarDadosTela(jogo);
            renderizarGameplay(dados, sprites, tickAnim);
            precisaRedesenhar = false;
        }

        char tecla = lerTecla(50);
        if (tecla != '\0') {
            processarTecla(jogo, tecla);
            precisaRedesenhar = true;
        }
    }

    desativarTerminal();
    return jogo.pontuacao;
}

void escrever_relatorio(std::ostream& out, const Metricas& m, const char* algoritmo){

    auto flags = out.flags();
    auto prec  = out.precision();

import TAD_lista_contigua;

export{
    enum class CriterioOrdenacao {
        POR_PONTOS,
        POR_NOME
    };

    //criterio me diz se a ordenação é por pontuação ou por nome
    void merge(ListaContigua_Historico& lista, ListaContigua_Historico& aux, int inicio, int meio, int fim, CriterioOrdenacao criterio) {
        for (int t = inicio; t <= fim; ++t) {
            aux.elementos[t] = lista.elementos[t];
        }

        int i = inicio;
        int j = meio + 1;
        int k = inicio;

        if (criterio == CriterioOrdenacao::POR_PONTOS) { // decrescente por pontuação
            while (i <= meio && j <= fim) {
                if (aux.elementos[i].pontuacao >= aux.elementos[j].pontuacao) {
                    lista.elementos[k++] = aux.elementos[i++];
                } else {
                    lista.elementos[k++] = aux.elementos[j++];
                }
            }
        } else { //ordenar por nome
            while (i <= meio && j <= fim) {
                if (aux.elementos[i].nome <= aux.elementos[j].nome) {
                    lista.elementos[k++] = aux.elementos[i++];
                } else {
                    lista.elementos[k++] = aux.elementos[j++];
                }
            }
        }
        
        while (i <= meio) {
            lista.elementos[k++] = aux.elementos[i++];
        }
        while (j <= fim) {
            lista.elementos[k++] = aux.elementos[j++];
        }
    }

    void merge_rec(ListaContigua_Historico& lista, ListaContigua_Historico& aux, int inicio, int fim, CriterioOrdenacao criterio) {
        if(inicio >= fim) {
            return;
        }
        int meio = inicio + (fim - inicio) / 2;
        merge_rec(lista, aux, inicio, meio, criterio);
        merge_rec(lista, aux, meio + 1, fim, criterio);
        merge(lista, aux, inicio, meio, fim, criterio);
    }

    void merge_sort(ListaContigua_Historico& lista, CriterioOrdenacao criterio) {
        if (lista.tamanho <= 1) {
            return;
        }

        ListaContigua_Historico aux;
        aux.tamanho = lista.tamanho;
        aux.capacidade = lista.capacidade;
        aux.elementos = std::make_unique<NodoListaContigua[]>(aux.capacidade);

        merge_rec(lista, aux, 0, lista.tamanho - 1, criterio);
    }

int larguraVisual(const string& s) {
    int colunas = 0;
    bool emAnsi = false;

    for (unsigned char c : s) {
        if (c == '\033') {
            emAnsi = true;
            continue;
        }
        if (emAnsi) {
            if (c == 'm') emAnsi = false;
            continue;
        }
        // Ignora bytes de continuação UTF-8 (10xxxxxx)
        if ((c & 0xC0) != 0x80) {
            colunas++;
        }
    }
    return colunas;
}

string ajustarColuna(const string& linha, int larguraAlvo) {
    int faltam = larguraAlvo - larguraVisual(linha);
    string out = linha + "\033[0m";
    if (faltam > 0) {
        out.append(static_cast<size_t>(faltam), ' ');
    }
    return out;
}

string decodificarAnsi(const string& linha) {
    string out;
    out.reserve(linha.size());
    for (size_t i = 0; i < linha.size(); ++i) {
        if (i + 3 < linha.size() && linha.compare(i, 4, "\\033") == 0) {
            out += '\033';
            i += 3;
        } else {
            out += linha[i];
        }
    }
    return out;
}

Frame juntarLadoALado(const Frame& esq, const Frame& dir, int espacoEntre = 3) {
    Frame resultado;
    resultado.altura = (esq.altura > dir.altura) ? esq.altura : dir.altura;
    if (resultado.altura > Frame::MAX_LINHAS) {
        resultado.altura = Frame::MAX_LINHAS;
    }
    resultado.largura = esq.largura + espacoEntre + dir.largura;

    string separador(static_cast<size_t>(espacoEntre), ' ');

    for (int i = 0; i < resultado.altura; ++i) {
        string lEsq = (i < esq.altura) ? esq.linhas[i] : "";
        string lDir = (i < dir.altura) ? dir.linhas[i] : "";

        resultado.linhas[i] = ajustarColuna(lEsq, esq.largura)
                            + separador
                            + ajustarColuna(lDir, dir.largura);
    }
    return resultado;
}

bool carregarAnimacao(const string& caminho, Animacao& anim) {
    std::ifstream arq(caminho);
    if (!arq.is_open()) return false;

    anim.totalFrames = 0;
    bool lendoFrame = false;
    string linha;

    while (std::getline(arq, linha)) {
        if (!linha.empty() && linha.back() == '\r') linha.pop_back();

        if (linha == "@FRAME") {
            if (anim.totalFrames >= Animacao::MAX_FRAMES) break;
            lendoFrame = true;
            anim.frames[anim.totalFrames] = Frame{};
            continue;
        }
        if (linha == "@END") {
            if (lendoFrame) {
                anim.totalFrames++;
                lendoFrame = false;
            }
            continue;
        }

        if (lendoFrame) {
            Frame& f = anim.frames[anim.totalFrames];
            if (f.altura < Frame::MAX_LINHAS) {
                string dec = decodificarAnsi(linha);
                int w = larguraVisual(dec);
                if (w > f.largura) f.largura = w;
                f.linhas[f.altura++] = dec;
            }
        }
    }
    return anim.totalFrames > 0;
}

        m.ord_tempo_nome = tempo;
    } else {
        m.ord_tempo_pontos = tempo;
    }

    // exibição (fora do cronômetro): só as primeiras N entradas
    const int N = 20;
    for (int i = 0; i < copia.tamanho && i < N; ++i) {
        std::cout << (i + 1) << ". " << copia.elementos[i].nome
                  << " - " << copia.elementos[i].pontuacao << "\n";
    }
    if (copia.tamanho > N) {
        std::cout << "... (" << (copia.tamanho - N) << " entradas omitidas)\n";
    }
    std::cout << "Tempo de ordenacao (Merge Sort): " << tempo << " s\n";

    gravar_log(m);   // regrava o medicoes.log com o bloco atualizado
} else if (opcao != 0) {
    std::cout << "Opcao invalida.\n";
}
    } while (opcao != 0);
}

static OpcaoMenu mostrar_menu(const PacoteSprites& sprites) {
    ativarTerminal();

    Timer timerAnim;
    timer_iniciar(&timerAnim);
    int tickAnim = 0;
    bool precisaRedesenhar = true;

    OpcaoMenu escolha = OpcaoMenu::SAIR;
    bool escolheu = false;

    while (!escolheu) {
        if (timer_parar(&timerAnim) >= 0.25) {
            tickAnim++;
            timer_iniciar(&timerAnim);
            precisaRedesenhar = true;
        }

        if (precisaRedesenhar) {
            renderizarLogo(sprites, tickAnim);
            precisaRedesenhar = false;
        }

        char tecla = lerTecla(50);
        switch (tecla) {
            case '1': escolha = OpcaoMenu::NOVO_JOGO;  escolheu = true; break;
            case '2': escolha = OpcaoMenu::COMO_JOGAR; escolheu = true; break;
            case '3': escolha = OpcaoMenu::HISTORICO;  escolheu = true; break;
            case '0':
            case 27:  escolha = OpcaoMenu::SAIR;       escolheu = true; break;
            default:  break;
        }
    }

    desativarTerminal();
    std::cout << "\033[2J\033[H" << std::flush;
    return escolha;

static std::string ler_nome_jogador() {

            return "Jogador";

        while (!nome.empty() && (nome.back() == '\r' || nome.back() == ' ' || nome.back() == '\t')) {

    // 1) Argumentos
    Argumentos args;
    if (!tratar_argumentos(argc, argv, args)) return 1;

    // 2) Estruturas principais
    Avl_Pedidos             avl;
    HashCache_Pedidos       cache;
    ListaContigua_Historico historico;
    inicializar_listacontigua(historico);

    // 3) Carga dos arquivos e sprites
    auto resultado_pedidos = carregar_pedidos(args.arquivo_pedidos, avl);
    if (!resultado_pedidos.sucesso) {
        std::cerr << "Erro ao carregar pedidos.\n";
        return 1;
    }

    auto resultado_historico = carregar_historico(args.arquivo_historico, historico);
    if (resultado_historico.ignoradas > 0) {
        std::cerr << "Aviso: " << resultado_historico.ignoradas
                  << " linha(s) do historico foram ignoradas por estarem malformadas.\n";

    g_metricas.avl_tempo_carga = resultado_pedidos.tempo_carregamento;
    coletar_metricas_carga(avl, historico, g_metricas);
    if (!gravar_log(g_metricas)) {

    }

    // 5) Loop do menu inicial
    bool rodando = true;
    while (rodando) {
        OpcaoMenu opcao = mostrar_menu(sprites);

                mostrar_como_jogar();
                break;

            case OpcaoMenu::HISTORICO:

                break;

            case OpcaoMenu::NOVO_JOGO: {

                avl.total_comparacao = 0;

            case OpcaoMenu::SAIR:
                rodando = false;
                break;
        }
    }

    return 0;
}

#include "timer.hpp"

void timer_iniciar (Timer* t) {
    t -> inicio = std :: chrono :: steady_clock :: now () ;
}

double timer_parar (Timer* t) {
    auto fim = std :: chrono :: steady_clock :: now () ;
    return std :: chrono :: duration < double >( fim - t -> inicio ). count () ;
}
